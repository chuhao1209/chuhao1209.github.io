我的分析与解答：

先考虑树的情况。

我们记录 dp(u,0) 表示这个点选择，这个以 u 为根子树的最大贡献。

dp(u,1) 表示这个点不选择，但是这个点的一个儿子是选择的，其它的点都没有选择的情况下以 u 为根的子树的最大贡献。

dp(u,2) 表示这个点不选择，并且这个点的儿子也不选择时，以 u 为根的子树的最大贡献。

对于 dp(u,0) 的转移，我们从 dp(v,2) 转移上来即可。

对于 dp(u,2) 的转移，我们从 max(dp(v,1),dp(v,2)) 转移上来即可。

而 dp(u,1)则较为复杂，我们要求只有一个儿子选择，那么可以通过以下式子计算：

dp(u,1)=dp(u,2)+max(dp(v,0)−max(dp(v,1),dp(v,2)))

综上，我们已经完成了树的情况。

再考虑环，我们把一个环上节点连向环外的树的贡献跑出来，将其计入我们的这个环上节点中，这可以在 tarjan 中转移实现。

然后我们发现环以后，我们先将整个环跑出来，然后将这个环转化成路径问题来处理。

我们希望把结果转移到节点 t 并且在得到结果后由 t 上传，那么我们就枚举 t 的状态来转移。

注意：以下分类的情况最为重要，也是被 hack 的原因所在。

设环的长度为 cnt 并且设 calc(l,r) 为 l−1 和 r+1 不选择而 l∼r 可以随便选择的情况下最大的贡献。

先考虑最困难的 dp(u,1)，因为这个需要从 dp(u,2) 转移，所以要提前处理。

假设旁边的点都不选时，那么只有选择的点在树上选择了，直接转移即可。

如果选择了旁边的点，那么这个点的树上点没有选择，枚举选择点在左边和在右边的情况。

枚举左右两边的情况时，需要去特判 cnt=3 的情况，否则计算中间值时会出错。

dp(u,0)的情况同理，也需要分类讨论，再进行转移。

dp(u,2)则因为计算的范围较长，不会出错，所以直接计算即可。

具体计算过程看代码即可，这里不过多描述了。

我的代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,ans,dp[1000005][3],a[1000005],pre[1000005],dfn[1000005],low[1000005],h[1000005],cnt,id,f[1000005][3];
vector<int> e[1000005];
int read(){
	char c=getchar();
	int x=0;
	while(c>'9'||c<'0')c=getchar();
	while(c>='0'&&c<='9')x=(x<<3)+(x<<1)+(c^48),c=getchar();
	return x;
}
int calc(int l,int r){
	f[l-1][0]=0,f[l-1][1]=dp[h[l-1]][1],f[l-1][2]=dp[h[l-1]][2];
	for(int i=l;i<=r+1;i++){
		f[i][0]=f[i-1][2]+dp[h[i]][0];
		f[i][2]=max(f[i-1][2],f[i-1][1])+dp[h[i]][2];
		f[i][1]=max(f[i-1][0]+dp[h[i]][2],max(f[i-1][2],f[i-1][1])+dp[h[i]][1]);
	}
	return max(f[r+1][1],f[r+1][2]);
}
void solve(int s,int t){
	cnt=0;
	int w=s;
	while(w!=t)h[++cnt]=w,w=pre[w];
	h[++cnt]=w;
	if(cnt>3)dp[t][1]=max(dp[t][1]+calc(2,cnt-2),dp[t][2]+max(dp[h[1]][0]+dp[h[2]][2]+calc(4,cnt-2),dp[h[cnt-1]][0]+dp[h[cnt-2]][2]+calc(2,cnt-4)));
	else if(cnt==3)dp[t][1]=max(dp[t][1]+calc(2,cnt-2),dp[t][2]+max(dp[h[1]][0]+dp[h[2]][2],dp[h[2]][0]+dp[h[1]][2]));
	if(cnt>3)dp[t][0]+=calc(3,cnt-3)+dp[h[1]][2]+dp[h[cnt-1]][2];
	else if(cnt==3)dp[t][0]+=dp[h[1]][2]+dp[h[3]][2];
	dp[t][2]+=calc(2,cnt-2);
}
void tarjan(int p,int fa){
	pre[p]=fa;dp[p][0]=a[p];dp[p][2]=0;
	dfn[p]=low[p]=++id;
	int ma=0;
	for(int i:e[p]){
		if(i==fa)continue;
		if(!dfn[i]){
			tarjan(i,p);
			low[p]=min(low[p],low[i]);
		}
		else low[p]=min(low[p],dfn[i]);
		if(low[i]>dfn[p]){
			dp[p][0]+=dp[i][2];
			dp[p][2]+=max(dp[i][2],dp[i][1]);
			ma=max(ma,dp[i][0]-max(dp[i][2],dp[i][1]));
		}
	}
	dp[p][1]=dp[p][2]+ma;
	for(int i:e[p])if(pre[i]!=p&&low[i]==dfn[p])solve(i,p);
}
signed main(){
	n=read(),m=read();
	for(int i=1;i<=n;i++)a[i]=read();
	for(int i=1,u,v;i<=m;i++){
		u=read(),v=read();
		e[u].push_back(v);
		e[v].push_back(u);
	}
	for(int i=1;i<=n;i++){
		if(!dfn[i]){
			tarjan(i,0);
			ans+=max({dp[i][0],dp[i][1],dp[i][2]});
		}
	}
	cout<<ans;
	return 0;
}
